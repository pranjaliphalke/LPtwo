import heapq
from collections import defaultdict

def prim_mst(graph, start=0):
    mst = []
    visited = set()
    min_heap = [(0, start, -1)]  # (weight, to_node, from_node)

    while min_heap:
        weight, to_node, from_node = heapq.heappop(min_heap)
        if to_node in visited:
            continue
        visited.add(to_node)
        if from_node != -1:
            mst.append((from_node, to_node, weight))
        for neighbor, edge_weight in graph[to_node]:
            if neighbor not in visited:
                heapq.heappush(min_heap, (edge_weight, neighbor, to_node))
    return mst

def main():
    graph = defaultdict(list)

    # Get number of vertices and edges
    n = int(input("Enter number of vertices: "))
    e = int(input("Enter number of edges: "))

    print("Enter edges in format: u v w (for edge from u to v with weight w)")
    for _ in range(e):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))  # undirected graph

    start_node = int(input("Enter starting node for MST: "))

    mst = prim_mst(graph, start=start_node)
    total_weight = sum(w for _, _, w in mst)

    print("\nMinimum Spanning Tree:")
    for u, v, w in mst:
        print(f"{u} - {v} : {w}")
    print(f"Total weight of MST: {total_weight}")

if __name__ == "__main__":
    main()
