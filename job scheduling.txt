profit = [15, 27, 10, 100, 150]
jobs = ["j1", "j2", "j3", "j4", "j5"]
deadline = [2, 3, 3, 3, 4]

# Combine, sort by profit descending
profitNJobs = list(zip(profit, jobs, deadline))
profitNJobs = sorted(profitNJobs, key=lambda x: x[0], reverse=True)

n = len(jobs)
slot = [0] * (n + 1)  # Use n+1 to accommodate index up to n
ans = ['null'] * (n + 1)

total_profit = 0

for i in range(n):
    job = profitNJobs[i]
    # Try to find a free slot from job deadline to 1
    for j in range(min(n, job[2]), 0, -1):
        if slot[j] == 0:
            slot[j] = 1
            ans[j] = job[1]
            total_profit += job[0]
            break

# Remove the 0th element which is unused
print("Jobs scheduled buddy:", ans[1:])
print("Total Profit:", total_profit)
